cmake_minimum_required(VERSION 3.20)
project(DRE VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DRE_DISPLAY "Build SDL2 software laboratory" ON)
add_library(dre_core src/core.cpp)
target_include_directories(dre_core PUBLIC include)
target_compile_options(dre_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dre_bench src/bench.cpp)
target_link_libraries(dre_bench PRIVATE dre_core)
if(DRE_DISPLAY)
  find_package(SDL2 REQUIRED)
  add_executable(dre_lab src/lab.cpp)
  target_link_libraries(dre_lab PRIVATE dre_core SDL2::SDL2)
endif()
enable_testing()
add_executable(dre_tests tests/core.cpp)
target_link_libraries(dre_tests PRIVATE dre_core)
add_test(NAME core COMMAND dre_tests)
add_test(NAME benchmark COMMAND dre_bench)
